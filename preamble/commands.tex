\newcommand{\setsecret}[2]{%
  \expandafter\def\csname secretdata@#1\endcsname{#2}%
}

\newcommand{\redactbox}[1][8em]{%
  \begingroup
  \setlength{\fboxsep}{0pt}%
  \colorbox{black}{\rule{0pt}{1.5ex}\hspace{#1}}%
  \endgroup
}

\newcommand{\secretwidth}[1]{%
  \@ifundefined{secretwidth@#1}
  {8em}
  {\csname secretwidth@#1\endcsname}%
}

\newcommand{\secret}[1]{%
  \@ifundefined{secretdata@#1}
  {\expandafter\redactbox\expandafter[\secretwidth{#1}]}
  {\csname secretdata@#1\endcsname}%
}

\newcommand{\emailentry}{%
  \@ifundefined{secretdata@email}
  {\small \redactbox[14em]}
  {\href{mailto:\csname secretdata@email\endcsname}{\small \csname
  secretdata@email\endcsname}}%
}
